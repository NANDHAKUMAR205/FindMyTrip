const axios = require('axios');

const getFlights = async () => {
  const url = 'https://liteapi.travel';
  
  // Define your flight itinerary structure
  const payload = {
    legs: [
      {
        origin: 'JFK',        // 3-letter IATA Airport code (e.g., John F. Kennedy International)
        destination: 'LAX',   // 3-letter IATA Airport code (e.g., Los Angeles International)
        date: '2026-11-15'    // Departure date format: YYYY-MM-DD
      }
    ],
    passengers: {
      adults: 1
    }
  };

  try {
    const response = await axios.post(url, payload, {
      headers: {
        'accept': 'application/json',
        'content-type': 'application/json',
        'X-API-Key': 'YOUR_SANDBOX_PUBLIC_KEY' // Replace with your Nuitée Sandbox Key
      }
    });

    console.log('Flight Offers Found:');
    console.dir(response.data, { depth: null }); // Prints the deep pricing and flight metrics cleanly
  } catch (error) {
    if (error.response) {
      console.error('API Error Response:', error.response.data);
    } else {
      console.error('Request Execution Error:', error.message);
    }
  }
};

getFlights();



const axios = require('axios');

const getHotels = async () => {
  try {
    const response = await axios.get('https://api.liteapi.travel/v3.0/data/hotels', {
      params: {
        countryCode: 'US',
        cityName: 'New York'
      },
      headers: {
        'accept': 'application/json',
        'X-API-Key': 'YOUR_SANDBOX_PUBLIC_KEY' // Paste your sandbox key here
      }
    });
    console.log(response.data);
  } catch (error) {
    console.error('Error fetching hotels:', error);
  }
};

getHotels();
